function openPage(page,el){ document.querySelectorAll(".page") .forEach(p=>p.classList.remove("show")); document.getElementById(page) .classList.add("show"); document.querySelectorAll(".navbtn") .forEach(b=>b.classList.remove("active")); el.classList.add("active"); if(page==="stocking") renderProducts(); if(page==="products") renderPriceList(); } function changeLocation(){ state.location= document.getElementById("locationSelect").value; document.getElementById("stockLocation").innerText= "Standort: "+state.location; updateDashboard(); } function renderProducts(){ const list=document.getElementById("productList"); list.innerHTML=""; state.products.forEach((p,i)=>{ list.innerHTML+=`
${p.name}
EK € ${p.price.toFixed(2)}
−
${p.qty}
+
`; }); updateStockTotal(); } function plus(i){ state.products[i].qty++; document.getElementById("q"+i).innerText= state.products[i].qty; updateStockTotal(); } function minus(i){ if(state.products[i].qty===0)return; state.products[i].qty--; document.getElementById("q"+i).innerText= state.products[i].qty; updateStockTotal(); } function updateStockTotal(){ let total=0; state.products.forEach(p=>{ total+=p.qty*p.price; }); document.getElementById("stockTotal") .innerText=euro(total); } function saveStocking(){ let items=[]; let total=0; state.products.forEach(p=>{ if(p.qty>0){ items.push({ name:p.name, qty:p.qty, price:p.price }); total+=p.qty*p.price; } }); if(items.length===0){ alert("Keine Produkte gewählt"); return; } state.stockings.push({ location:state.location, items, total, date:new Date().toISOString() }); state.products.forEach(p=>p.qty=0); renderProducts(); updateDashboard(); alert("Wareneinschlichtung gespeichert"); } function saveCashout(){ const coins= Number(document.getElementById("coins").value); const notes= Number(document.getElementById("notes").value); const total=coins+notes; state.cashouts.push({ location:state.location, coins, notes, total, date:new Date().toISOString() }); document.getElementById("coins").value=""; document.getElementById("notes").value=""; updateDashboard(); alert("Entleerung gespeichert"); } function renderPriceList(){ const list=document.getElementById("priceList"); list.innerHTML=""; state.products.forEach((p,i)=>{ list.innerHTML+=`
${p.name}
`; }); } function changePrice(i,v){ state.products[i].price= Number(v); } function updateDashboard(){ let cash= state.cashouts.filter(x=>{ return state.location==="Gesamt" ||x.location===state.location; }); let stock= state.stockings.filter(x=>{ return state.location==="Gesamt" ||x.location===state.location; }); const revenue= cash.reduce((a,b)=>a+b.total,0); const stockValue= stock.reduce((a,b)=>a+b.total,0); const profit= revenue-stockValue; const month= new Date().getMonth(); const monthRevenue= cash.filter(x=>{ return new Date(x.date).getMonth()===month; }) .reduce((a,b)=>a+b.total,0); document.getElementById("kpiRevenue") .innerText=euro(revenue); document.getElementById("kpiProfit") .innerText=euro(profit); document.getElementById("kpiStock") .innerText=euro(stockValue); document.getElementById("kpiMonth") .innerText=euro(monthRevenue); renderHistory(); } function renderHistory(){ const div= document.getElementById("historyList"); div.innerHTML=""; let all=[]; state.cashouts.forEach(x=>{ all.push({ type:"Entleerung", value:x.total, location:x.location, date:x.date }); }); state.stockings.forEach(x=>{ all.push({ type:"Einschlichtung", value:x.total, location:x.location, date:x.date }); }); all=all.filter(x=>{ return state.location==="Gesamt" ||x.location===state.location; }); all.sort((a,b)=>new Date(b.date)-new Date(a.date)); if(all.length===0){ div.innerHTML="Noch keine Einträge"; return; } all.forEach(x=>{ div.innerHTML+=`
${x.type}
${x.location}

${euro(x.value)}

${new Date(x.date).toLocaleString("de-DE")}
`; }); } function runReport(){ const from= new Date(document.getElementById("from").value); const to= new Date(document.getElementById("to").value); let cash= state.cashouts.filter(x=>{ const d=new Date(x.date); return d>=from&&d<=to; }); let stock= state.stockings.filter(x=>{ const d=new Date(x.date); return d>=from&&d<=to; }); if(state.location!=="Gesamt"){ cash=cash.filter(x=>x.location===state.location); stock=stock.filter(x=>x.location===state.location); } const revenue= cash.reduce((a,b)=>a+b.total,0); const stockValue= stock.reduce((a,b)=>a+b.total,0); const profit= revenue-stockValue; document.getElementById("repRevenue") .innerText=euro(revenue); document.getElementById("repStock") .innerText=euro(stockValue); document.getElementById("repProfit") .innerText=euro(profit); } changeLocation(); renderProducts(); renderPriceList(); updateDashboard();
